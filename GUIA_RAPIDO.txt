📝 Como integrar no index.js:

1. No topo do index.js, adicione:
   require('./comandos/meu_primeiro_comando')(bot);
   const { meuPrimeiroComando } = require('./comandos/meu_primeiro_comando');

2. Dentro de bot.on('callback_query', ...):
   if (query.data === 'meucomando') await meuPrimeiroComando(bot, chatId);

3. Adicione um botão no arquivo comandos_buttons.js:
   { text: '👋 Meu Comando', callback_data: 'meucomando' }

📂 Arquivo: comandos/pergunta.js

────────────────────────────────────────────────────────────────────────────
// No index.js, crie um Set para rastrear quem está respondendo:
const aguardandoResposta = new Set();

// Função do comando:
async function pergunta(bot, chatId) {
    aguardandoResposta.add(chatId);
    await bot.sendMessage(chatId, '❓ Qual é seu nome?');
}

// No bot.on('message'):
bot.on('message', async (msg) => {
    const chatId = msg.chat.id;
    
    if (aguardandoResposta.has(chatId)) {
        aguardandoResposta.delete(chatId);
        const nome = msg.text;
        await bot.sendMessage(chatId, 
            `👋 Prazer conhecê-lo, ${nome}!`
        );
    }
});

module.exports = { pergunta };

const ADMINS = [123456789, 987654321];

async function comandoAdmin(bot, chatId, userId) {
    if (!ADMINS.includes(userId)) {
        await bot.sendMessage(chatId, 
            '❌ Você não tem permissão para usar este comando!'
        );
        return;
    }
    
    await bot.sendMessage(chatId, 
        '✅ Comando de admin executado com sucesso!'
    );
}

module.exports = { comandoAdmin };
